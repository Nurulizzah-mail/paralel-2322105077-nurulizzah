Perbedaan Implementasi Kode di C (OpenMP)

Kode Aman (Wajib Digunakan)

#pragma omp parallel num_threads(threads) reduction(+:total_inside)
{
    xor128_state rng; // RNG privat per thread
    #pragma omp for schedule(static)
    for (uint64_t i = 0; i < N; i++) {
        if (x*x + y*y <= 1.0) {
            total_inside++; // AMAN: Masuk register lokal thread
        }
    }
}

Kode Berbahaya (Race Condition)

volatile uint64_t unsafe_inside = 0;
#pragma omp parallel num_threads(threads)
{
    #pragma omp for schedule(static)
    for (uint64_t i = 0; i < N; i++) {
        if (x*x + y*y <= 1.0) {
            unsafe_inside++; // RUSAK: 4 thread menimpa memori bersamaan!
        }
    }
}
